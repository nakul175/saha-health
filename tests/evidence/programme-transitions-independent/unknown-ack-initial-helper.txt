import {writeFileSync,mkdirSync} from 'node:fs';
import {transitionPhone} from './programme-transition-phone.mjs';
export async function transitionPhoneIndependent(args){
 await transitionPhone(args);const {base,owner,org,ownerName,fromId,toId,out,check}=args;
 const {chromium}=await import(process.env.SAHA_BROWSER_PACKAGE??'@playwright/test'),browser=await chromium.launch({headless:true,...(process.env.SAHA_CHROMIUM?{executablePath:process.env.SAHA_CHROMIUM}:{})});let posts=0,lookups=0,request=null,saved=null;
 try{const c=await browser.newContext({viewport:{width:1200,height:900},extraHTTPHeaders:{...owner.headers,'x-saha-organisation':org}}),p=await c.newPage();await p.goto(base+'/');await p.getByText(ownerName,{exact:true}).first().waitFor();await p.getByRole('button',{name:'Organisation & team',exact:true}).click();await p.setViewportSize({width:390,height:844});const panel=p.locator('section').filter({has:p.getByRole('heading',{name:'Reviewed programme transitions',exact:true})}).last();await panel.getByRole('button',{name:'Load transition plans',exact:true}).click();await panel.getByLabel('Original installed version').selectOption(fromId);await panel.getByLabel('Target installed version').selectOption(toId);await panel.getByLabel('Effective time',{exact:true}).fill(new Date(Date.now()+10800000).toISOString().slice(0,16));await panel.getByLabel('Transition reason').fill('Synthetic independent lost-response transition evidence');
 await p.route('**/api/installed-pack-transition*',async route=>{const r=route.request();if(r.method()==='POST'){posts++;request={body:r.postData(),key:r.headers()['idempotency-key'],expectedActor:r.headers()['x-saha-expected-actor']};const response=await route.fetch();saved={status:response.status(),body:await response.json()};await route.abort('failed');}else{if(new URL(r.url()).searchParams.has('receiptKey'))lookups++;await route.continue();}});
 await panel.getByRole('button',{name:'Stage exact transition'}).click();await panel.getByText('Response unconfirmed. Do not repeat or create another operation.',{exact:true}).waitFor();check(posts===1&&saved.status===200,'Real accepted transition transport loss has one POST');check(request.expectedActor===owner.headers['oai-authenticated-user-id'],'Original owner bound to transport before mutation');check(await panel.getByRole('button',{name:'Stage exact transition'}).isDisabled(),'Unknown response prevents another stage');check(await panel.getByText('Original transition reference: '+request.key,{exact:true}).isVisible(),'Phone retains original nonce after lost acknowledgement');
 await panel.getByRole('button',{name:'Check original transition result'}).click();await panel.getByText('Original transition reference: '+request.key,{exact:true}).waitFor({state:'hidden'});check(posts===1&&lookups===1,'Read-only receipt recovery makes no second POST or new nonce');check(await panel.getByRole('heading',{name:'Team default · transition staged',exact:true}).count()===1,'Recovered accepted stage is visible once, not independently approved/applied');check(await panel.getByRole('button',{name:'Stage exact transition'}).isEnabled(),'Fresh same-actor read recovers governed controls');
 const outcome=await c.request.get(base+'/api/installed-pack-transition?receiptKey='+request.key);const j=await outcome.json();check(outcome.status()===200&&j.outcome==='accepted'&&j.result.transitionId===saved.body.result.transitionId,'Current receipt is exact original durable accepted outcome');await panel.screenshot({path:out+'/phone/independent-lost-ack-recovered.png'});writeFileSync(out+'/independent-phone-lost-ack.json',JSON.stringify({posts,lookups,request,saved,receipt:j},null,2)+'\n');await c.close();
 }finally{await browser.close();}
}
