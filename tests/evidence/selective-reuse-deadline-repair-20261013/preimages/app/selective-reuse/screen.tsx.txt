 'use client';
import {useCallback,useEffect,useRef,useState} from 'react';
import {selectiveReuseContext,type SelectiveReuseContext} from '@/lib/saha/selective-reuse';
import './style.css';
const visible=()=>document.visibilityState!=='hidden';
export function SelectiveReuseScreen(){const[context,setContext]=useState<SelectiveReuseContext|null>(null),[message,setMessage]=useState('Loading current record authority…'),[busy,setBusy]=useState(false),[visitId,setVisit]=useState(''),[destinationId,setDestination]=useState(''),[purpose,setPurpose]=useState('');const org=useRef(''),person=useRef(''),actor=useRef(''),selection=useRef({visitId:'',destinationId:'',purpose:''}),epoch=useRef(0),controller=useRef<AbortController|null>(null);
 const clear=useCallback(()=>{epoch.current++;controller.current?.abort();setContext(null);setBusy(false);},[]);
 const load=useCallback(async(choice=selection.current)=>{clear();selection.current=choice;setVisit(choice.visitId);setDestination(choice.destinationId);setPurpose(choice.purpose);if(!navigator.onLine||!visible()){setMessage('Offline or hidden: protected payload cleared.');return;}const token=epoch.current,abort=new AbortController();controller.current=abort;setBusy(true);setMessage('Checking this exact destination and purpose against current authority…');try{const q=new URLSearchParams({personId:person.current});if(choice.visitId)q.set('visitId',choice.visitId);if(choice.destinationId&&choice.purpose){q.set('destinationId',choice.destinationId);q.set('purpose',choice.purpose);}const r=await fetch('/api/selective-reuse?'+q,{cache:'no-store',signal:abort.signal,headers:{'x-saha-organisation':org.current,...(actor.current?{'x-saha-operation-actor':actor.current}:{})}});if(!r.ok)throw Error('Unavailable');const value=selectiveReuseContext.parse(await r.json());if(value.organisationId!==org.current||value.personId!==person.current||actor.current&&value.actorId!==actor.current||r.headers.get('x-saha-operation-actor')!==value.actorId||r.headers.get('x-saha-organisation')!==value.organisationId||value.refreshAt&&Date.parse(value.refreshAt)<=Date.now()||choice.destinationId&&choice.purpose&&(value.selected?.visitId!==choice.visitId||value.selected?.destinationId!==choice.destinationId||value.selected?.purpose!==choice.purpose))throw Error('Unbound');if(token!==epoch.current||!visible()||!navigator.onLine)return;actor.current=value.actorId;setContext(value);setMessage(value.selected?.state==='withheld'?'Reuse withheld: no fact values or old payload are shown.':value.selected?'Current selected mapping preview only; nothing was sent.':'Choose an existing confirmed encounter and one destination/purpose.');}catch{if(token===epoch.current)setMessage('Reuse review unavailable. Protected payload cleared; no action or disclosure was performed.');}finally{if(token===epoch.current)setBusy(false);}},[clear]);
 useEffect(()=>{const q=new URLSearchParams(window.location.search);if(q.size!==2||q.getAll('personId').length!==1||q.getAll('organisationId').length!==1||!q.get('personId')||!q.get('organisationId')){queueMicrotask(()=>setMessage('Open from one explicit organisation and existing person.'));return;}org.current=q.get('organisationId')!;person.current=q.get('personId')!;let alive=true;queueMicrotask(()=>{if(alive)void load();});const lose=()=>{clear();setMessage('Protected payload cleared. Refresh current authority before reviewing again.');},offline=()=>{clear();setMessage('Offline: protected payload cleared.');},hidden=()=>{if(!visible())lose();};window.addEventListener('offline',offline);window.addEventListener('pagehide',lose);document.addEventListener('visibilitychange',hidden);return()=>{alive=false;clear();window.removeEventListener('offline',offline);window.removeEventListener('pagehide',lose);document.removeEventListener('visibilitychange',hidden);};},[clear,load]);
 useEffect(()=>{if(!context?.refreshAt)return;const timer=setTimeout(()=>{clear();setMessage('Assignment deadline reached. Protected payload cleared.');},Math.max(0,Math.min(Date.parse(context.refreshAt)-Date.now(),2147483647)));return()=>clearTimeout(timer);},[context,clear]);
 const s=context?.selected;return <main className="reuse"><h1>Review permitted data reuse</h1><p>Permission for one destination or purpose does not authorise another. This screen checks existing configured mappings and current grants; it creates no plan, approval, submission or AI request.</p><p role="status">{message}</p><button disabled={busy} onClick={()=>void load()}>Refresh current authority</button><button disabled={busy} onClick={()=>void load({visitId:'',destinationId:'',purpose:''})}>Clear selection</button>{context&&<><label>Existing confirmed encounter<select value={visitId} disabled={busy} onChange={e=>void load({visitId:e.target.value,destinationId:'',purpose:''})}><option value="">Choose a confirmed encounter</option>{context.visits.map(v=><option key={v.id} value={v.id}>{v.date} · version {v.version} · {v.id}</option>)}</select></label><label>Existing destination<select value={destinationId} disabled={busy||!visitId} onChange={e=>void load({visitId,destinationId:e.target.value,purpose:e.target.value?purpose:''})}><option value="">Choose one destination</option>{context.destinations.map(d=><option key={d.id} value={d.id}>{d.name} · {d.kind} · version {d.version}</option>)}</select></label><label>Original purpose<select value={purpose} disabled={busy||!visitId||!destinationId} onChange={e=>void load({visitId,destinationId,purpose:e.target.value})}><option value="">Choose one purpose</option><option value="programme-delivery">Programme delivery (MIS)</option><option value="ai-assistance">AI assistance</option></select></label>
 {s&&<section aria-label="Selected reuse review"><h2>{s.state==='withheld'?'Reuse withheld':'Permitted mapping preview'}</h2><p>Exact destination {s.destinationId}; purpose {s.purpose}; encounter version {s.visitVersion}.</p>{s.reason&&<p>{s.reason}</p>}{s.payload&&<><h3>Selected destination fields only</h3><dl>{Object.entries(s.payload.fields).map(([target,value])=><div key={target}><dt>{target}</dt><dd>{value===null?'Explicit null (configured mapping)':value===''?'Explicit empty (configured mapping)':value}</dd></div>)}</dl><h3>Original configured field allowlist</h3><ul>{s.mapping.map(m=><li key={m.target}>{m.source} → {m.target}; {m.required?'required':'optional'}; transformation {m.transformation??'identity'}; unit {m.unit??'original default'}; missing {m.nullSemantics??'omit'}</li>)}</ul><p>Not mapped to this destination (field names only): {s.excluded.join(', ')}.</p><p className="hash">Selected payload digest {s.payloadHash}</p>{s.originalPlan?<p>Original own prepare receipt matches this current selected payload. Retained plan {s.originalPlan.id}; recorded state {s.originalPlan.state}. Approval and sending remain separate.</p>:<p>No original own prepared-plan receipt can currently be matched to this selected payload. A stored plan elsewhere is not approval for this use.</p>}</>}{s.state==='withheld'&&<p>All mapped fact values, stored payloads and original plan results are withheld. Reuse elsewhere grants no permission here.</p>}</section>}
 <p>Only current original mapped programme measurements are previewed. Free-text notes and correction history require their original wider privacy/history process. AI payloads require original capability and minimisation review; none is built here. Configured allowlists do not prove partner, clinical or domain approval.</p></>}</main>;}
