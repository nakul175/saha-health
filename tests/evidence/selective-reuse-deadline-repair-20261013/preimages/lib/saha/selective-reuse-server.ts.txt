import {identity,db,json,runtime} from './server';
import {readState,scope} from './store';
import {fail} from './core';
import {environmentOf} from './environment';
import {patientId} from './patient-directory';
import {recordAccess} from './access-audit';
import {reuseSelection,selectiveReuseProjection} from './selective-reuse';
import {selectiveReuseTransferProof,SELECTIVE_REUSE_TRANSFER_SQL} from './selective-reuse-read-guard';
const specs=[['org','SELECT * FROM organisations WHERE id=?'],['members','SELECT * FROM memberships WHERE org_id=? ORDER BY user_id'],['teams','SELECT * FROM team_members WHERE org_id=? ORDER BY user_id,team_id'],['ops','SELECT * FROM operations WHERE org_id=?'],['governance','SELECT * FROM governance WHERE org_id=?'],['integrations','SELECT * FROM integrations WHERE org_id=? ORDER BY id'],['person','SELECT * FROM patients WHERE org_id=? AND id=?'],['visits','SELECT * FROM visits WHERE org_id=? AND person_id=? ORDER BY id'],['submissions','SELECT * FROM submissions WHERE org_id=? AND person_id=? ORDER BY id'],['plans','SELECT * FROM plans WHERE org_id=? AND visit_id IN (SELECT id FROM visits WHERE org_id=? AND person_id=?) ORDER BY id'],['ai','SELECT * FROM ai_results WHERE org_id=? AND visit_id IN (SELECT id FROM visits WHERE org_id=? AND person_id=?)'],['receipts',"SELECT actor_id,org_id,person_id,key,hash,command_type,result FROM command_receipts WHERE org_id=? AND person_id=? AND command_type='preparePlan' ORDER BY actor_id,key"],['transfers',SELECTIVE_REUSE_TRANSFER_SQL]]as const;
async function proof(org:string,id:string){const rows=await db().batch(specs.map(([n,sql])=>db().prepare(sql).bind(...(['plans','ai'].includes(n)?[org,org,id]:['person','visits','submissions','receipts'].includes(n)?[org,id]:[org]))));return Object.fromEntries(specs.map(([n],i)=>[n,rows[i].results]))as Record<typeof specs[number][0],Record<string,string|number|null>[]>;}
export async function selectiveReuseRead(request:Request){
 if(!['synthetic','training'].includes(environmentOf(runtime()).name))fail(409,'Use a synthetic or training reuse review.');const actor=await identity(),org=patientId(request.headers.get('x-saha-organisation')),expected=request.headers.get('x-saha-operation-actor');if(expected!==null&&expected!==actor.userId)fail(403,'The original signed-in record account is required.');let parsed;try{parsed=reuseSelection(new URL(request.url).searchParams);}catch{fail(400,'Choose exact current person, encounter, destination and original purpose.');}const{id,selected}={id:parsed!.personId,selected:parsed!.selected};const before=await proof(org,id),state=await readState(db(),org,[id],actor.userId,true),member=scope(state,actor.userId);
 const patient=before.person[0],loaded=state.workspace.people.find(p=>p.id===id);if(!loaded||!patient||patient.id!==loaded.id||patient.version!==loaded.version||patient.team_id!==loaded.teamId||patient.assigned_to!==loaded.assignedTo||JSON.stringify({...JSON.parse(patient.data as string),version:patient.version,teamId:patient.team_id,assignedTo:patient.assigned_to,assigned:loaded.assigned})!==JSON.stringify(loaded))fail(404,'Current selected record source is unavailable.');
 if(before.integrations.length!==state.workspace.integrations.length||before.integrations.some(r=>{const i=state.workspace.integrations.find(i=>i.id===r.id);return!i||JSON.stringify(JSON.parse(r.data as string))!==JSON.stringify(i)||i.id!==r.id||i.version!==r.version||i.status!==r.status||i.kind!==r.kind;}))fail(409,'Current destination source changed.');
 for(const [name,source]of [['ops',state.operations],['governance',state.governance]]as const)if(JSON.stringify(before[name][0]?JSON.parse(before[name][0].data as string):undefined)!==JSON.stringify(source))fail(409,'Current policy source changed.');
 if(!before.members.some(r=>r.user_id===member.userId&&r.version===member.version&&r.role===member.role&&r.status===member.status&&!!r.is_owner===member.isOwner)||JSON.stringify(before.teams.filter(r=>r.user_id===member.userId).map(r=>r.team_id).sort())!==JSON.stringify([...member.teamIds].sort()))fail(409,'Current membership source changed.');
 if(before.visits.length!==state.workspace.visits.length||before.visits.some(r=>{const v=state.workspace.visits.find(v=>v.id===r.id);return!v||JSON.stringify((()=>{const{planId,...data}=JSON.parse(r.data as string);return{...data,version:r.version,recordVersion:r.record_version,plan:planId?JSON.parse(String(before.plans.find(p=>p.id===planId)?.data??'null')):undefined,aiResults:before.ai.filter(a=>a.visit_id===r.id).map(a=>JSON.parse(a.data as string))};})())!==JSON.stringify(v)||v.personId!==id;}))fail(409,'Current encounter source changed.');if(JSON.stringify(before.submissions.map(r=>JSON.parse(r.data as string)))!==JSON.stringify(state.workspace.submissions))fail(409,'Current delivery history source changed.');
 const transfer=selectiveReuseTransferProof(before.transfers,new Date().toISOString()),value=await selectiveReuseProjection(state,member,actor.userId,id,selected,before.receipts,transfer.refreshAt);await recordAccess(db(),state,actor,member,{action:'viewActionPlan',subject:id,personId:id,detail:'Selective destination/purpose mapped payload review, current allowlist only; no new plan, disclosure, provider or submission.'});
 const after=await proof(org,id);if(JSON.stringify(after)!==JSON.stringify(before))fail(409,'Authority, permission, mapping, source or original receipt changed before review.');selectiveReuseTransferProof(after.transfers,new Date().toISOString());if(value.refreshAt&&Date.parse(value.refreshAt)<=Date.now())fail(409,'Current assignment deadline reached.');const r=json(value);r.headers.set('x-saha-operation-actor',actor.userId);r.headers.set('x-saha-organisation',org);return r;
}
