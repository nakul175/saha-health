'use client';
import {useCallback,useEffect,useRef,useState} from 'react';import type {TodayRow} from '@/lib/saha/today';import {assistanceSchema,cancellationAttempt,cancelledMatches,requestLabel,type AssistanceContext,type CancellationAttempt} from '@/lib/saha/assistance-control';import './assistance-control.css';
type View={kind:'loading'|'error';message:string}|{kind:'list';records:TodayRow[]}|{kind:'record';value:AssistanceContext};
export default function AssistanceControl(){const [view,setView]=useState<View>({kind:'loading',message:'Loading assigned records…'}),[chosen,setChosen]=useState(''),[reviewed,setReviewed]=useState(false),[attempt,setAttempt]=useState<CancellationAttempt|null>(null),[busy,setBusy]=useState(false),[notice,setNotice]=useState('');const active=useRef<AbortController|null>(null),org=useRef<string|null>(null),actor=useRef<string|null>(null),writing=useRef(false);
 const load=useCallback(async(personId?:string,visitId?:string)=>{if(writing.current)return;active.current?.abort();const controller=new AbortController();active.current=controller;const expectedOrg=personId?org.current:null,expectedActor=personId?actor.current:null;if(!personId){org.current=null;actor.current=null;}setChosen('');setReviewed(false);setView({kind:'loading',message:'Loading current authorised assistance records…'});if(!navigator.onLine||personId&&!expectedOrg){setView({kind:'error',message:'Assistance status unavailable. Reconnect and refresh assigned records.'});return;}try{const response=await fetch(personId?'/api/assistance-control?personId='+encodeURIComponent(personId)+(visitId?'&visitId='+encodeURIComponent(visitId):''):'/api/today',{cache:'no-store',credentials:'same-origin',signal:controller.signal,headers:expectedOrg?{'x-saha-organisation':expectedOrg}:{}});if(controller.signal.aborted)return;if(!response.ok)throw Error([401,403,404].includes(response.status)?'Access refused. Check your current ASHA membership and assignment.':'Assistance status unavailable. Refresh with a connection.');const actualOrg=response.headers.get('x-saha-organisation');if(!actualOrg||expectedOrg&&actualOrg!==expectedOrg)throw Error('Selected organisation could not be matched.');if(personId){const value=assistanceSchema.parse(await response.json()),actualActor=response.headers.get('x-saha-operation-actor');if(controller.signal.aborted)return;if(!actualActor||value.actorId!==actualActor||expectedActor&&expectedActor!==actualActor||value.person.id!==personId||value.organisationId!==expectedOrg||visitId&&value.selected?.id!==visitId)throw Error('Selected account or request context could not be matched.');actor.current=actualActor;setView({kind:'record',value});}else{const body=await response.json() as {records:TodayRow[]};if(controller.signal.aborted)return;if(!Array.isArray(body.records))throw Error('Assigned records unavailable.');org.current=actualOrg;setView({kind:'list',records:body.records});}}catch(cause){if(!controller.signal.aborted)setView({kind:'error',message:cause instanceof Error?cause.message:'Assistance status unavailable.'});}},[]);
 useEffect(()=>{const frame=requestAnimationFrame(()=>void load());const clear=()=>{active.current?.abort();org.current=null;actor.current=null;setChosen('');setReviewed(false);setView({kind:'error',message:'Protected assistance status cleared. Reconnect and refresh assigned records.'});};const visibility=()=>{if(document.hidden)clear();};window.addEventListener('offline',clear);window.addEventListener('pagehide',clear);document.addEventListener('visibilitychange',visibility);return()=>{cancelAnimationFrame(frame);active.current?.abort();window.removeEventListener('offline',clear);window.removeEventListener('pagehide',clear);document.removeEventListener('visibilitychange',visibility);};},[load]);
 const submit=async(original:CancellationAttempt)=>{if(writing.current)return;writing.current=true;setBusy(true);setAttempt(original);setChosen('');setReviewed(false);setNotice('Cancellation outcome unconfirmed here. Keep this original attempt reference.');active.current?.abort();const controller=new AbortController();active.current=controller;setView({kind:'loading',message:'Stopping future local application for the reviewed request…'});try{const response=await fetch('/api/assistance-control',{method:'POST',cache:'no-store',credentials:'same-origin',signal:controller.signal,headers:{'Content-Type':'application/json','Idempotency-Key':original.key,'x-saha-organisation':original.organisationId,'x-saha-operation-actor':original.actorId},body:JSON.stringify({personId:original.personId,command:original.command})});if(controller.signal.aborted)return;if(!response.ok)throw Error('Cancellation outcome not confirmed. Refresh authorised status or recover only this original attempt.');const body=await response.json() as {outcome:string;key:string;context:unknown};const current=assistanceSchema.parse(body.context);if(controller.signal.aborted)return;if(response.headers.get('x-saha-operation-actor')!==original.actorId||response.headers.get('x-saha-organisation')!==original.organisationId||body.outcome!=='cancelled-locally'||body.key!==original.key||!cancelledMatches(original,current))throw Error('Cancellation could not be matched to the original account and request.');org.current=original.organisationId;actor.current=original.actorId;setView({kind:'record',value:current});setAttempt(null);setNotice('Local cancellation recorded and read back. Already-sent requests or accepted external actions were not recalled.');}catch(cause){if(!controller.signal.aborted)setView({kind:'error',message:cause instanceof Error?cause.message:'Cancellation outcome unconfirmed.'});}finally{writing.current=false;setBusy(false);}};
 const start=()=>{if(view.kind!=='record'||!reviewed||attempt)return;try{void submit(cancellationAttempt(view.value,chosen,crypto.randomUUID()));}catch(cause){setNotice(cause instanceof Error?cause.message:'Review a current request first.');}};const value=view.kind==='record'?view.value:null;
 return <main className="assistance-control" aria-busy={busy||view.kind==='loading'}><p>Saha · Synthetic / training</p><h1>Assistance status and stop</h1><p>This screen starts no assistance request or provider call. Stop blocks future local use of an open result. It cannot recall a sent request, undo saved facts or reverse an externally accepted action.</p>{notice&&<p role="status">{notice}</p>}{attempt&&<section aria-label="Original cancellation attempt"><p>Original cancellation reference: {attempt.key}</p><p>Outcome unconfirmed here. Only the original account and exact attempt can recover it. Reload does not retain this in-memory reference.</p><button disabled={busy||view.kind==='loading'} onClick={()=>void submit(attempt)}>Recover original cancellation attempt</button></section>}
 {view.kind==='list'&&<section><h2>Choose an assigned record</h2>{!view.records.length&&<p>No assigned records.</p>}<ol>{view.records.map(p=><li key={p.id}><h3>{p.name}</h3><p>{p.village} · {p.id}</p><button aria-label={'Review assistance '+p.id} onClick={()=>void load(p.id)}>Review assistance status</button></li>)}</ol></section>}
 {value&&<section aria-label="Selected assistance record"><h2>{value.person.name}</h2><p>{value.person.id} · {value.person.village}</p><p>Only this account's requests are shown. Latest 50 current local visits are offered; this is not a complete archive or a record of all external activity.</p>{!value.visits.length&&<p>No current local visits found. This read creates no visit.</p>}<ul>{value.visits.map(v=><li key={v.id}><p>{v.date||'Observation time not recorded'} · {v.state} · {v.id}</p><button disabled={busy} aria-label={'Open assistance visit '+v.id} onClick={()=>void load(value.person.id,v.id)}>Open this visit's requests</button></li>)}</ul>
 {value.selected&&<section aria-label="Selected assistance visit"><h3>Visit {value.selected.id}</h3><p>Clinical version {value.selected.version} · Recorded version {value.selected.recordVersion}</p>{!value.selected.requests.length&&<p>No assistance request records for this account and selected visit. This does not prove no external work exists.</p>}{value.selected.requests.map(r=><article key={r.id}><h4>{requestLabel(r)} · {r.task}</h4><p>Request {r.id} · {r.provider.name} · {r.provider.mode==='local'?'Rules only; no language model':r.provider.mode+' recorded provider'} · Version {r.provider.version}</p><p>Created {r.createdAt} · Unsent expiry {r.expiresAt}</p><p>Captured time limit: {r.limits?.timeMinutes?r.limits.timeMinutes+' minutes':'Not recorded'}. Captured attempt limit: {r.limits?.attempts??'Not recorded'}. Default unsent expiry is at most {value.defaultExpiryMinutes} minutes.</p><p>Recorded request attempts: {r.callCount}. An attempt may reserve work before processing; this counter does not prove delivery. This screen performs no provider retry.</p><details><summary>Recorded request references</summary><p>Provider version {r.provider.version} · Captured visit {r.visitVersion}, person {r.personVersion}, permission {r.consentVersion}</p><p>Captured capability: {r.governance?r.governance.id+' · Version '+r.governance.version+' · '+r.governance.manifestHash:'Not recorded; no capability approval is inferred'}</p><p>Cost configuration version {value.costConfigurationVersion}. Other tool/step bounds are not recorded by this view.</p></details><p>Request cost estimate: {r.estimate?'USD '+r.estimate.usd.toFixed(4)+' · '+r.estimate.basis+(r.estimate.ratesAt?' · Rates dated '+r.estimate.ratesAt:''):'Unavailable; usage or cost assumptions are not established'}. {r.provider.mode==='local'?'Rules-only request; this excludes hosting and other charges.':'Reserved estimates use '+value.reservedTokens+' tokens; not a provider bill or known total spend.'}</p><p>Version-matched configured monthly limit: {r.monthlyUsd===null?'Unavailable or not recorded':'USD '+r.monthlyUsd.toFixed(2)}. No remaining balance or total actual spend is inferred.</p><ul>{r.steps.map((step,i)=><li key={i}>{step.name} · {step.state}</li>)}</ul>{(r.status==='running'||r.outcome==='uncertain'||r.callCount>0)&&<p>Provider outcome may be unresolved. Local cancellation never proves the request was not processed.</p>}{['pending','running','review'].includes(r.status)&&!attempt&&<button disabled={busy} aria-pressed={chosen===r.id} aria-label={'Choose request to stop '+r.id} onClick={()=>{setChosen(r.id);setReviewed(false);}}>Choose this request to stop</button>}</article>)}</section>}
 {chosen&&!attempt&&<fieldset disabled={busy}><legend>Explicit local stop for {chosen}</legend><button aria-pressed={reviewed} onClick={()=>setReviewed(v=>!v)}>{reviewed?'Local cancellation limits reviewed':'I reviewed the request and understand sent work is not recalled'}</button><button disabled={!reviewed} onClick={start}>Stop future local application</button><button onClick={()=>{setChosen('');setReviewed(false);}}>Cancel stop selection</button></fieldset>}<button disabled={busy} onClick={()=>void load(value.person.id,value.selected?.id)}>Refresh selected assistance status</button></section>}
 {(view.kind==='loading'||view.kind==='error')&&<p role="status">{view.message}</p>}<button disabled={busy||view.kind==='loading'} onClick={()=>void load()}>Refresh assigned records</button><a href="/today">Return to Today</a></main>;
}
