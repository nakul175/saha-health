"use client";
import {useCallback,useEffect,useRef,useState} from 'react';import {ownCarePermissionsSchema,type OwnCarePermissions} from '@/lib/saha/own-care-permissions';import {ACCESS_PURPOSE_LABEL} from '@/lib/saha/purpose-scope';import './style.css';
export default function Screen(){const [org,setOrg]=useState<string|null>(null),[view,setView]=useState<OwnCarePermissions|null>(null),[busy,setBusy]=useState(false),[notice,setNotice]=useState('Checking your own care context…');const epoch=useRef(0),actor=useRef<string|null>(null),request=useRef<AbortController|null>(null);
 const clear=useCallback((message:string)=>{epoch.current++;request.current?.abort();setView(null);setBusy(false);setNotice(message);},[]);
 const load=useCallback(async(id:string)=>{const e=++epoch.current;request.current?.abort();const c=new AbortController();request.current=c;setView(null);setBusy(true);setNotice('Checking current recorded restrictions…');try{if(!navigator.onLine)throw Error('Offline: reconnect and refresh your current care context.');const r=await fetch('/api/own-care-permissions',{cache:'no-store',redirect:'manual',signal:c.signal,headers:{'x-saha-organisation':id,...(actor.current?{'x-saha-operation-actor':actor.current}:{})}});if(!r.ok)throw Error('Current care restrictions unavailable ('+r.status+').');const parsed=ownCarePermissionsSchema.safeParse(await r.json());if(e!==epoch.current||document.visibilityState==='hidden'||!navigator.onLine)return;if(!parsed.success)throw Error('Current care source could not be matched.');const v=parsed.data;if(v.organisationId!==id||v.actorId!==r.headers.get('x-saha-operation-actor')||actor.current&&actor.current!==v.actorId||v.refreshAt&&Date.parse(v.refreshAt)<=Date.now())throw Error('Current care account or assignment context changed.');actor.current=v.actorId;setView(v);setNotice('Your recorded restrictions were checked. Each actual record or action still needs current server authorization.');}catch(error){if(e===epoch.current)setNotice(error instanceof Error?error.message:'Current care context unavailable.');}finally{if(e===epoch.current)setBusy(false);}},[]);
 useEffect(()=>{const q=new URLSearchParams(location.search),id=q.get('organisationId'),frame=requestAnimationFrame(()=>{if(q.getAll('organisationId').length!==1||[...q.keys()].some(k=>k!=='organisationId')||!id||!/^[A-Za-z0-9_:.-]{1,120}$/.test(id)){clear('Open your own care permissions from Today in the current organisation.');return;}setOrg(id);void load(id);});return()=>{cancelAnimationFrame(frame);epoch.current++;request.current?.abort();};},[clear,load]);
 useEffect(()=>{const offline=()=>clear('Offline: recorded restrictions cleared. Reconnect and refresh.'),hidden=()=>{if(document.visibilityState==='hidden')clear('Hidden: recorded restrictions cleared. Refresh current context.');},left=()=>clear('Page left: recorded restrictions cleared.');window.addEventListener('offline',offline);document.addEventListener('visibilitychange',hidden);window.addEventListener('pagehide',left);return()=>{window.removeEventListener('offline',offline);document.removeEventListener('visibilitychange',hidden);window.removeEventListener('pagehide',left);};},[clear]);
 useEffect(()=>{if(!view?.refreshAt)return;const id=setTimeout(()=>clear('Assignment context deadline reached. Refresh before continuing.'),Math.max(0,Math.min(Date.parse(view.refreshAt)-Date.now(),2147483647)));return()=>clearTimeout(id);},[view,clear]);
 return <main className="care-permissions"><h1>My care permissions</h1><p>Recorded restrictions for your current ASHA account. This screen cannot grant access or approve a record, export or tool call.</p><p role="status">{notice}</p>{org&&<button disabled={busy} onClick={()=>void load(org)}>Refresh my care permissions</button>}{view&&<section aria-label="My recorded care restrictions"><h2>Assigned care</h2><p>Current active ASHA membership · version {view.membershipVersion}.</p><p>Records require your direct assignment and current team scope.</p><ul>{view.teams.map(t=><li key={t.id}>{t.name} · {t.id}</li>)}</ul>{!view.teams.length&&<p>No current team is recorded; this is not permission to open any record.</p>}<h2>Programme narrowing</h2><p>{view.programmeRestriction??'No programme narrowing is recorded by your current purpose grant.'}</p><h2>Purpose narrowing</h2>{view.purposeRestriction?<ul>{view.purposeRestriction.map(p=><li key={p}>{ACCESS_PURPOSE_LABEL[p]}</li>)}</ul>:<p>No purpose narrowing is recorded by this grant. Each action retains its own server checks.</p>}{(['facility','locality']as const).map(key=><div key={key}><h2>{key==='facility'?'Facility':'Geography'} constraints</h2><p>{view[key].enforced?'Configured enforcement is on.':'Configured enforcement is off; team, assignment and other boundaries still apply.'}</p><p>These are recorded references, not validated directory or clinical approval.</p><ul>{view[key].references.map(x=><li key={x.id}>{x.name??'Reference name unavailable'} · {x.id}</li>)}</ul>{!view[key].references.length&&<p>No {key==='facility'?'facility':'locality'} reference is recorded for your current teams/account.{view[key].enforced&&' An enforced dimension without a matching reference cannot widen record access.'}</p>}</div>)}<p>Recorded location policy version {view.scopeVersion}. Checked {view.asOf}.</p></section>}<a href="/today">Return to assigned records</a><p>Permissions can change. The server rechecks current tenant, role, assignment, programme, facility, geography, purpose and action for each request.</p></main>;
}
