function pins(directory, exclusions = new Set(['node_modules', '.git', '.wrangler', 'reports', 'evidence'])) {
  const out = [];
  const walk = (path, relative) => { for (const entry of readdirSync(path, { withFileTypes: true })) {
    if (exclusions.has(entry.name)) continue;
    const next = join(path, entry.name), rel = join(relative, entry.name);
    if (entry.isDirectory()) walk(next, rel);
    else if (entry.isFile()) { const bytes = readFileSync(next); out.push({ path: rel, bytes: bytes.length, sha256: hash(bytes) }); }
  } };
  walk(directory, '');
  return out;
}

async function parsedResponse(response, personId, kind) {
  const body = await response.json(), headers = response.headers();
  assert.equal(response.status(), 200, JSON.stringify(body)); assert.equal(headers['cache-control'], 'no-store');
  if (kind === 'list') { assert.equal(headers['x-saha-organisation'], fixture.org); assert.deepEqual(body.records, fixture.exactAssignedRows); }
  else if (kind === 'detail') { assert.equal(headers['x-saha-organisation'], fixture.org); assert.deepEqual(body.record, fixture.exactSelectedRecord); }
  else {
    assert.equal(headers['x-saha-operation-actor'], fixture.asha.id);
    assert.deepEqual(body.person, fixture.serverPersonBefore); assert.equal(body.visit.id, fixture.visitId); assert.equal(body.organisationId, fixture.org);
    assert.equal(body.visit.version, fixture.serverVisitVersionBefore); assert.deepEqual(body.visit.facts, fixture.serverFactsBefore);
  }
  return { status: response.status(), cacheControl: headers['cache-control'], organisationBinding: kind === 'routine' ? body.organisationId : headers['x-saha-organisation'],
    trustedActorBinding: kind === 'routine' ? headers['x-saha-operation-actor'] : null, bytes: Buffer.byteLength(JSON.stringify(body)), sha256: hash(JSON.stringify(body)), exactPersonId: personId,
    visitId: kind === 'routine' ? fixture.visitId : null, version: kind === 'routine' ? body.visit.version : null };
}

        } catch (error) {

            error: String(error), declaredFaults: declaredFaults.slice(faultIndex), declaredExpectedTransportFault: declaredFaults.slice(faultIndex).some(fault => fault.sample?.setup === true) };
          setupFailures.push(failure);
          appendFileSync(join(folder, 'EXCLUDED-NATIVE-SETUP-RECOVERY.jsonl'), JSON.stringify(failure) + '\n');
          if (attempt === 2) throw error;
        }
      }
    }

          } else if (phase === 'connected-today-detail') {

            assert.equal(new URL(href, worker.base).searchParams.get('organisationId'), fixture.org);

        } catch (error) {
          journeyHadError = true;

          const elapsed = phaseStart === null ? null : performance.now() - phaseStart;
          writeRecord({ path: phase, sample: n, profile: profile.id, outcome: phaseStart === null ? 'setup-error-not-entered' : 'error', error: String(error), observedElapsedMs: elapsed,
            timeout: /timeout|timed out|within20000ms/i.test(String(error)), declaredExpectedTransportFault: faults.length > 0, actualDeclaredFaults: faults,
            completedStageTimes: times, actualBodies: bodies, dom: await page.locator('body').innerText().catch(() => 'not readable') });
        }
      }
      const wholeMs = performance.now() - wholeStart;
      writeRecord({ path: 'connected-whole-journey', sample: n, profile: profile.id, outcome: journeyHadError ? 'error' : 'measured', ms: journeyHadError ? null : wholeMs,
        observedElapsedMs: wholeMs, targetMs: null, targetExceeded: null, stageTimes: times,

    }

  }

}
function distribution(values) {
  const ordered = values.filter(value => Number.isFinite(value) && value >= 0).sort((a,b) => a-b);
  const percentile = proportion => ordered.length ? ordered[Math.ceil(proportion*ordered.length)-1] : null;
  return { n: ordered.length, median: percentile(.5), p95: percentile(.95), p99: percentile(.99), max: ordered.at(-1) ?? null };
}
function plannedPaths() {
  const combinations = [];
  if (mode !== 'connected') for (const path of ['autosave','explicit-save','cached-patient','cached-today']) combinations.push(['warm-desktop-offline-browser',path]);
  if (mode !== 'device') for (const profile of profiles.filter(profile => selectedProfileMode === 'both' || !profile.abortEvery))
    for (const path of ['connected-today-list','connected-today-detail','connected-routine-entry']) combinations.push([profile.id,path]);

}
function summarise() {
  const combinations = [...plannedPaths(), ...profiles.map(profile => [profile.id,'connected-whole-journey']).filter(([profile,path]) => records.some(record => record.profile === profile && record.path === path))];
  return combinations.map(([profile,path]) => {
    const selected = records.filter(record => record.profile === profile && record.path === path), successes = selected.filter(record => record.outcome === 'measured');
    const errors = selected.filter(record => record.outcome !== 'measured');
    const allObserved = selected.map(record => record.outcome === 'measured' ? record.ms : record.observedElapsedMs).filter(Number.isFinite);
    const completionDistribution = distribution(successes.map(record => record.ms)), allAttemptObservedStopDistribution = distribution(allObserved);
    const ceiling = thresholds[path] ?? null;
    const exceeded = ceiling === null ? [] : selected.filter(record => (record.outcome === 'measured' ? record.ms : record.observedElapsedMs) > ceiling);
    return { profile,path,plannedFixedAttempts:samplesPerPath,attemptsRecorded:selected.length,completedScreens:successes.length,
      errors:errors.length,declaredExpectedTransportErrors:errors.filter(record => record.declaredExpectedTransportFault).length,
      unexplainedErrors:errors.filter(record => !record.declaredExpectedTransportFault).length,timeouts:errors.filter(record => record.timeout).length,
      notEntered:errors.filter(record => record.outcome === 'setup-error-not-entered').length,missingAttemptRecords:Math.max(0,samplesPerPath-selected.length),
      missingObservedIntervals:selected.length-allObserved.length,successfulSamplesAtLeast1000:successes.length>=1000,
      allAttemptObservedStopDistribution,successCompletionDistribution:completionDistribution,
      allAttemptDistributionMeaning:'Stop/failure intervals include actual failed attempts; failed-fast is not a completed-screen timing.',
      originalNumericalCeilingMs:ceiling,
      observedCompletionP95WithinNumericalCeiling:ceiling === null || !completionDistribution.n ? null : completionDistribution.p95<=ceiling,
      observedAllAttemptStopP95WithinNumericalCeiling:ceiling === null || !allAttemptObservedStopDistribution.n ? null : allAttemptObservedStopDistribution.p95<=ceiling,
      cachedCompletedEntriesOver2s:path.startsWith('cached-')?successes.filter(record => record.ms>2000).length:null,
      observedCeilingExceedances:exceeded.map(record => ({sample:record.sample,outcome:record.outcome,elapsedMs:record.outcome==='measured'?record.ms:record.observedElapsedMs})),
      formalPhysicalR1Acceptance:false };
  });
}
function disposition(summaries) {
  const findings=[];
  const unexplainedSetupFailures = setupFailures.filter(failure => !failure.declaredExpectedTransportFault);
  if (unexplainedSetupFailures.length) findings.push({ kind: 'engineering-control-unexplained-setup-errors', count: unexplainedSetupFailures.length,
    note: 'Recovered native setup failures remain failures; conservative control budget0 is separate from original formal NFR02.' });
  for (const summary of summaries.filter(summary => summary.path !== 'connected-whole-journey')) {
    const identity={profile:summary.profile,path:summary.path};
    if (summary.attemptsRecorded !== samplesPerPath || summary.missingObservedIntervals || summary.notEntered)
      findings.push({...identity,kind:'incomplete-measurement',expected:samplesPerPath,actual:summary.attemptsRecorded,missingObservedIntervals:summary.missingObservedIntervals,notEntered:summary.notEntered});
    if (summary.unexplainedErrors) findings.push({...identity,kind:'engineering-control-unexplained-errors',count:summary.unexplainedErrors,
      note:'Conservative local control budget0; not an original formal NFR zero-error rule.'});
    if (summary.path.startsWith('cached-') && summary.cachedCompletedEntriesOver2s)
      findings.push({...identity,kind:'original-cached2s-numerical-exceedance',count:summary.cachedCompletedEntriesOver2s});
    if (!summary.path.startsWith('cached-') && summary.successCompletionDistribution.p95 > summary.originalNumericalCeilingMs)
      findings.push({...identity,kind:'original500ms-or3s-completion-p95-numerical-exceedance',actual:summary.successCompletionDistribution.p95,ceiling:summary.originalNumericalCeilingMs});
    if (summary.allAttemptObservedStopDistribution.p95 > summary.originalNumericalCeilingMs)
      findings.push({...identity,kind:'engineering-observed-stop-p95-exceeds-numerical-ceiling',actual:summary.allAttemptObservedStopDistribution.p95,
        ceiling:summary.originalNumericalCeilingMs,note:'Failure stop distribution is not labelled completed-screen performance; no failed attempt was filtered away.'});
  }
  return findings;
}
