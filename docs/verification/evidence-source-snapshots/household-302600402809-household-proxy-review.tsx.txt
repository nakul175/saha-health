'use client';
import {useCallback,useEffect,useRef,useState} from 'react';
import {householdProxyResponse,type HouseholdProxyReview as Review} from '@/lib/saha/household-proxy-review';
import {PURPOSES} from '@/lib/saha/household-constants';
import './household-proxy-review.css';
export default function HouseholdProxyReview(){
 const[review,setReview]=useState<Review|null>(null),[notice,setNotice]=useState('Loading current relationships…'),[recipient,setRecipient]=useState(''),[purpose,setPurpose]=useState<typeof PURPOSES[number]>('care-coordination'),[loading,setLoading]=useState(false);const generation=useRef(0),pending=useRef<AbortController|null>(null),actor=useRef<string|null>(null),selection=useRef<{id:string;org:string}|null>(null);
 const invalidate=useCallback(()=>{generation.current++;pending.current?.abort();pending.current=null;},[]);
 const clear=useCallback((message:string)=>{invalidate();setReview(null);setRecipient('');setLoading(false);setNotice(message);},[invalidate]);
 const load=useCallback(async(recipientId?:string,sharingPurpose?:typeof PURPOSES[number])=>{pending.current?.abort();const n=++generation.current,controller=new AbortController();pending.current=controller;setReview(null);setLoading(true);setNotice('Loading current relationships…');const selected=selection.current;if(!selected){clear('Choose an assigned record from Today.');return;}if(document.hidden||!navigator.onLine){clear('Protected relationships cleared. Reconnect and refresh.');return;}
 try{const q=new URLSearchParams({personId:selected.id});if(recipientId&&sharingPurpose){q.set('recipientId',recipientId);q.set('purpose',sharingPurpose);}const response=await fetch('/api/household-proxy-review?'+q,{cache:'no-store',credentials:'same-origin',signal:controller.signal,headers:{'x-saha-organisation':selected.org,...(actor.current?{'x-saha-operation-actor':actor.current}:{})}});if(controller.signal.aborted||generation.current!==n||document.hidden||!navigator.onLine)return;if(!response.ok){clear([401,403,404].includes(response.status)?'Access refused. Return to Today and check your current membership.':'Relationships unavailable or changed. Refresh current access.');return;}const data=householdProxyResponse.parse(await response.json());if(controller.signal.aborted||generation.current!==n||document.hidden||!navigator.onLine)return;if(data.organisationId!==selected.org||data.person.id!==selected.id||response.headers.get('x-saha-organisation')!==selected.org||response.headers.get('x-saha-operation-actor')!==data.actorId||actor.current&&actor.current!==data.actorId||Boolean(recipientId)!==Boolean(data.preview)||data.preview&&(data.preview.recipientId!==recipientId||data.preview.purpose!==sharingPurpose)||data.refreshAt&&Date.parse(data.refreshAt)<=Date.now()){clear('Relationship context changed. Return to Today.');return;}actor.current=data.actorId;setReview(data);setRecipient(data.preview?.recipientId??'');setNotice(data.preview?'Permitted name preview loaded. No information was sent.':'Current worker relationship metadata loaded.');setLoading(false);
 }catch{if(!controller.signal.aborted&&generation.current===n)clear('Relationships unavailable. Refresh current access.');}},[clear]);
 useEffect(()=>{let cancelled=false;queueMicrotask(()=>{if(cancelled)return;const q=new URLSearchParams(location.search),id=q.get('personId'),org=q.get('organisationId'),valid=(x:string|null)=>typeof x==='string'&&/^[A-Za-z0-9_:.-]{1,100}$/.test(x);if([...q.keys()].some(k=>!['personId','organisationId'].includes(k))||q.getAll('personId').length!==1||q.getAll('organisationId').length!==1||!valid(id)||!valid(org)){clear('Choose an assigned record from Today.');return;}selection.current={id:id!,org:org!};void load();});return()=>{cancelled=true;invalidate();};},[clear,load,invalidate]);
 useEffect(()=>{const close=()=>clear('Protected relationships cleared. Reconnect and refresh.'),hidden=()=>{if(document.hidden)close();};addEventListener('offline',close);addEventListener('pagehide',close);document.addEventListener('visibilitychange',hidden);return()=>{removeEventListener('offline',close);removeEventListener('pagehide',close);document.removeEventListener('visibilitychange',hidden);};},[clear]);
 const refreshAt=review?.refreshAt;
 useEffect(()=>{if(!refreshAt)return;const delay=Math.max(0,Date.parse(refreshAt)-Date.now());const t=setTimeout(()=>clear('A recorded proxy interval changed. Refresh current relationships.'),Math.min(delay,2147483647));return()=>clearTimeout(t);},[refreshAt,clear]);
 const todayHref='/today';
 return <main className="household-proxy"><header><a href={todayHref}>Back to Today</a><h1>Household and proxy review</h1><p>Household membership and a family contact do not grant account access or permission to disclose another person’s history.</p></header><p role="status">{notice}</p><button disabled={loading} onClick={()=>void load()}>Refresh current relationships</button>{review&&<section aria-label="Current household relationships"><h2>{review.person.name}</h2>{!review.household?<p>No current household is recorded.</p>:<><p>{review.household.label} · recorded version {review.household.version}</p><p>Worker view: only people in your current care scope. This screen contains no private clinical history.</p>{review.limited&&<p>Other household relationships are outside this view. No names or counts are shown.</p>}<ul>{review.members.map(m=><li key={m.id}><h3>{m.name}</h3><p>Household relationship: {m.relationship} · recorded from {m.from}</p>{m.grants.length===0?<p>No proxy grants to a visible current household member are recorded.</p>:m.grants.map(g=><article key={g.id}><h4>Recorded proxy: {g.recipientName}</h4><p>Relationship: {g.relationship} · Status: {g.status}</p><p>Purposes: {g.purposes.join(', ')}</p><p>Sensitive classes: {g.classes.length?g.classes.join(', '):'None named'}</p><p>Recorded evidence: {g.evidence}</p><p>Starts {g.from} · Expires {g.expiresAt}{g.revokedAt?' · Revoked '+g.revokedAt:''}</p><p>Recorded evidence is not independent verification or account access.</p></article>)}</li>)}</ul><form onSubmit={e=>{e.preventDefault();void load(recipient,purpose);}}><h2>Review a recipient and purpose</h2><p>Choose explicitly. This local preview shows permitted names only; it does not send information or open any clinical history.</p><label>Current household recipient<select value={recipient} onChange={e=>setRecipient(e.target.value)}><option value="">Choose a recipient</option>{review.members.map(m=><option key={m.id} value={m.id}>{m.name} — {m.relationship}</option>)}</select></label><label>Disclosure purpose<select value={purpose} onChange={e=>setPurpose(e.target.value as typeof PURPOSES[number])}>{PURPOSES.map(x=><option key={x} value={x}>{x}</option>)}</select></label><button disabled={!recipient||loading}>Review permitted names</button></form>{review.preview&&<section aria-label="Permitted name preview"><h2>Names permitted for this recipient and purpose</h2><p>Recipient: {review.members.find(m=>m.id===review.preview!.recipientId)?.name} · {review.preview.purpose}</p><ul>{review.preview.subjects.map(s=><li key={s.id}>{s.name} — {s.self?'Recipient’s own record':'Current recorded proxy authority'} · {s.relationship}</li>)}</ul><p>People without a current matching proxy grant are omitted. Sensitive clinical details are not displayed or sent.</p></section>}</>}</section>}</main>;
}
